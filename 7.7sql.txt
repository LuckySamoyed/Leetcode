select 
    a.*,
    expid1
from 
    (select 
        day,
        imei,
        case when scene_app='1' then '主端'
            when scene_app='2' then '挂件'
            when scene_app='3' then '负一屏' end as scene_app,
        case when imei_source='2' then '头条'
            when imei_source='99' then '自研' else '其他' end as imei_source,
        case when is_feed_new_imei_14 =1 then '14日新用户' else '老用户' end as is_new_imei,
        case when brn_module='资讯' and channel_name in ('推荐','发现','要闻','视频','探索') then channel_name else '其他' end as channel_name,
        case when content_type in ('图文','横版','竖版') then content_type 
             when content_type in ('短内容','小说','信息流热榜新闻','铺底内容') then '其他泛图文'
             else '其他' end as content_type,
        case when brn_module='资讯' and out_side='列表页' then 1 
             when brn_module='资讯' then 0 else null end as is_liebiao,
        case when brn_module='资讯' and in_side<>'' then in_side 
             when brn_module='资讯' then '' else null end as in_side,
        case when click_cnt>0 then 1 else 0 end as is_click,
        expid,
        max(nvl(is_abnormal_imei,'0')) as is_abnormal_imei,
        max(is_feed_act_imei) as is_feed_act_imei,
        sum(show_cnt) as show_cnt, 
        sum(show_num) as show_num,
        sum(click_cnt) as click_cnt,
        sum(click_num) as click_num,
        sum(play_cnt) as play_cnt,
        sum(play_num) as play_num,
        sum(play_time) as play_time,
        sum(max_play_time) as max_play_time,
        sum(video_time) as video_time,
        sum(valid_play_num) as valid_play_num,
        sum(reading_rate) as reading_rate,
        sum(like_num) as like_num,
        sum(comment_num) as comment_num,
        sum(negative_num) as negative_num,
        sum(channel_duration) as channel_duration,
        sum(using_duration) as using_duration,
        count(distinct request_id) as request_alg_num,
        sum(collection_num) as collection_num,
        sum(share_num) as share_num,
        sum(interact_num) as interact_num
    from 
        bi_alg_dev.dm_brn_imei_news_info_di a
    where 
        day='2025-07-07' 
    group by 
        day,
        imei,
        scene_app,
        case when scene_app='1' then '主端'
            when scene_app='2' then '挂件'
            when scene_app='3' then '负一屏' end,
        case when imei_source='2' then '头条'
            when imei_source='99' then '自研' else '其他' end ,
        case when is_feed_new_imei_14 =1 then '14日新用户' else '老用户' end ,
        case when brn_module='资讯' and channel_name in ('推荐','发现','要闻','视频','探索') then channel_name else '其他' end,
        case when content_type in ('图文','横版','竖版') then content_type 
             when content_type in ('短内容','小说','信息流热榜新闻','铺底内容') then '其他泛图文'
             else '其他' end,
        case when brn_module='资讯' and out_side='列表页' then 1 
             when brn_module='资讯' then 0 else null end,
        case when brn_module='资讯' and in_side<>'' then in_side 
             when brn_module='资讯' then '' else null end,
        case when click_cnt>0 then 1 else 0 end,
